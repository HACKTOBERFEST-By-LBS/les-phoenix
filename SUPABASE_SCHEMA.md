## Schéma de base de données Supabase
## Exécuter dans : https://supabase.com/dashboard/project/ztoqkyjmhzdnltbnabjq/editor

```sql
-- Table des utilisateurs
CREATE TABLE IF NOT EXISTS users (
    pseudo      TEXT PRIMARY KEY,
    salt        TEXT NOT NULL,
    hash        TEXT NOT NULL,
    lang        TEXT DEFAULT 'fr',
    created_at  TIMESTAMPTZ DEFAULT NOW()
);

-- Historique des scans SMS
CREATE TABLE IF NOT EXISTS scan_history (
    id          BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    user_pseudo TEXT REFERENCES users(pseudo) ON DELETE CASCADE,
    timestamp   TIMESTAMPTZ DEFAULT NOW(),
    sms_preview TEXT,
    verdict     TEXT CHECK (verdict IN ('ok', 'douteux', 'arnaque')),
    amount      TEXT,
    sender      TEXT,
    advice      TEXT
);

-- Historique des cours
CREATE TABLE IF NOT EXISTS course_history (
    id           BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    user_pseudo  TEXT REFERENCES users(pseudo) ON DELETE CASCADE,
    timestamp    TIMESTAMPTZ DEFAULT NOW(),
    course_name  TEXT,
    last_command TEXT
);

-- Index de performance
CREATE INDEX IF NOT EXISTS idx_scan_user   ON scan_history(user_pseudo);
CREATE INDEX IF NOT EXISTS idx_course_user ON course_history(user_pseudo);

-- Désactiver RLS (backend utilise service_role)
ALTER TABLE users          DISABLE ROW LEVEL SECURITY;
ALTER TABLE scan_history   DISABLE ROW LEVEL SECURITY;
ALTER TABLE course_history DISABLE ROW LEVEL SECURITY;
```
